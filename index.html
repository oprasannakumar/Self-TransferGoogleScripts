function enableSwipeToDelete(wrapper) { if (!wrapper) return; const content = wrapper.querySelector('.mode-row-content'); const row = wrapper.closest('.mode-row'); const container = row.parentElement; let startX = 0, startY = 0; let currentX = 0; let isDragging = false; let isHorizontal = false; wrapper.addEventListener('touchstart', (e) => { if (e.target.closest('input, select, button')) return; const touch = e.touches[0]; startX = touch.clientX; startY = touch.clientY; isDragging = true; isHorizontal = false; content.style.transition = 'none'; }, { passive: true }); wrapper.addEventListener('touchmove', (e) => { if (!isDragging) return; const touch = e.touches[0]; const dx = touch.clientX - startX; const dy = touch.clientY - startY; // Decide direction (IMPORTANT) if (!isHorizontal) { if (Math.abs(dx) > 10) { isHorizontal = true; } else if (Math.abs(dy) > 10) { // user is scrolling → cancel swipe isDragging = false; return; } } if (isHorizontal) { if (dx < 0) { e.preventDefault(); // 🔥 critical currentX = dx; content.style.transform = `translateX(${dx}px)`; } } }, { passive: false }); wrapper.addEventListener('touchend', () => { if (!isDragging) return; isDragging = false; const rows = container.querySelectorAll('.mode-row'); const isFirstRow = rows[0] === row; if (currentX < -80 && !isFirstRow) { content.style.transition = '0.3s ease'; content.style.transform = 'translateX(-120%)'; content.style.opacity = '0'; setTimeout(() => { row.remove(); calcSplitTotals(); autoSyncDestination(); }, 300); } else { content.style.transition = '0.2s ease'; content.style.transform = 'translateX(0)'; } currentX = 0; }); }
